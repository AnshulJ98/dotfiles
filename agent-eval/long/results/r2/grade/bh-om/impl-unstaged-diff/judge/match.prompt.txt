+  --base   ref the slice started from (default HEAD); the working tree is compared with it

+function atBase(root: string, base: string, file: string): string | undefined {
+  try {
+    return git(root, ["show", `${base}:${file}`]);

+  }
+}

+function main(argv: string[]): number {
+  const split = argv.indexOf("--");
+  const command = split === -1 ? [] : argv.slice(split + 1);
+  let values: { base?: string; files?: string; help?: boolean };
+  try {
+    ({ values } = parseArgs({

+/**
+ * Pure checks behind bin/slice-check: whether a slice's diff stayed inside its card's files and left its
+ * tests at least as strong as it found them. lib/slice-check-cli.ts gathers the facts from git and the
+ * test command; everything here is a function of those facts.
+ */
+import { isAbsolute, matchesGlob, normalize } from "node:path";
+
+/** True when a repo-relative path matches one of the globs. `**` does not match dotfiles (node:path matchesGlob). */
+export function isAllowed(repoRelativePath: string, globs: readonly string[]): boolean {
+  if (!repoRelativePath || isAbsolute(repoRelativePath)) return false;
+  const path = normalize(repoRelativePath);
+  if (path.startsWith("..")) return false;
+  return globs.some((glob) => path === normalize(glob) || matchesGlob(path, glob));
+}
+
+export const TEST_FILE = /(\.(test|spec)\.[cm]?[jt]sx?$)|(^|\/)__tests__\/|(^|\/)test_[^/]*\.py$|_test\.py$/;
+const TEST_CASE = /\b(it|test|describe)\s*\(|\bassert\b|\bexpect\s*\(|\bdef\s+test_/g;
+const TEST_SKIP =
+  /\.(skip|only|todo)\s*\(|\b(xit|xdescribe|xtest|fit|fdescribe)\s*\(|\b(skip|only|todo)\s*:\s*true|@pytest\.mark\.(skip|xfail)\b/g;
+
+function occurrences(text: string | undefined, pattern: RegExp): number {
+  return text === undefined ? 0 : (text.match(pattern) ?? []).length;
+}
+
+/**
+ * Why a test file's change looks like the tests were weakened rather than the code fixed: fewer cases or
+ * assertions than before, or new skip/only markers. `undefined` content means the file does not exist;
+ * a new file has nothing to weaken.
+ */
+export function testWeakening(before: string | undefined, after: string | undefined): string | undefined {
+  if (before === undefined) return undefined;
+  const casesBefore = occurrences(before, TEST_CASE);
+  const casesAfter = occurrences(after, TEST_CASE);
+  if (casesAfter < casesBefore) return `test cases or assertions dropped from ${casesBefore} to ${casesAfter}`;
+  if (occurrences(after, TEST_SKIP) > occurrences(before, TEST_SKIP)) return "new skip/only/todo markers";
+  return undefined;
+}
+
+export interface TestFileChange {
+  file: string;
+  before: string | undefined;
+  after: string | undefined;
+}
+
+export interface SliceFacts {
+  /** Repo-relative paths changed since the base, untracked new files included. */
+  changed: readonly string[];
+  /** The card's files; undefined when none were given, which skips the scope check. */
+  allowed: readonly string[] | undefined;
+  testFiles: readonly TestFileChange[];
+  command: string;
+  exitCode: number | null;
+}
+
+/** Every reason the slice fails; empty means PASS. */
+export function sliceFailures(facts: SliceFacts): string[] {
+  const reasons: string[] = [];
+  if (facts.exitCode !== 0) reasons.push(`\`${facts.command}\` exited ${String(facts.exitCode)}`);
+  const allowed = facts.allowed;
+  if (allowed) {
+    const outside = facts.changed.filter((file) => !isAllowed(file, allowed));
+    if (outside.length) reasons.push(`files outside the card: ${outside.join(", ")}`);
+  }
+  for (const { file, before, after } of facts.testFiles) {
+    const why = testWeakening(before, after);
+    if (why) reasons.push(`${file}: ${why}`);
+  }
+  return reasons;
+}
