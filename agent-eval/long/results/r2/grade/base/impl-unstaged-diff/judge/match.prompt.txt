+ * slice-check: the end-of-slice check, run by /plan "go" or by hand.
+ *
+ *   slice-check <base> [--tests '<cmd>'] [--files '<glob>,<glob>']
+ *
+ * Compares the working tree (tracked changes plus untracked files) with
+ * <base> and fails when:
+ *   - the --tests command exits non-zero (it runs through sh);
+ *   - a changed test file has fewer test cases or assertions than at <base>,
+ *     or new skip/only/todo markers;
+ *   - a changed path matches none of the --files globs.
+ * Prints one `SLICE-CHECK: PASS|FAIL <reasons>` line last. Exit 0 pass,
+ * 1 fail, 2 usage error. To run it on every commit, call it from a
+ * pre-commit hook with the base and arguments the slice uses.
+ *
+ * `**` in --files does not match dotfiles (node:path matchesGlob); list
+ * them by name.
+ *
+ * Known defects: a renamed test file counts as deleted (no rename
+ * detection), so it fails as dropped cases; --tests has no timeout.
+ */

+import { readFileSync, statSync } from "node:fs";
+import { isAbsolute, join, matchesGlob, normalize } from "node:path";

+const TEST_FILE = /(\.(test|spec)\.[cm]?[jt]sx?$)|(^|\/)__tests__\/|(^|\/)test_[^/]*\.py$|_test\.(py|go)$/;
+const TEST_CASE = /\b(it|test|describe)\s*\(|\bassert\b|\bexpect\s*\(|^\s*def test_/gm;
+const TEST_SKIP = /\.(skip|only|todo)\s*\(|\b(xit|xdescribe|xtest|fit|fdescribe)\s*\(|\b(skip|only|todo)\s*:\s*true|@pytest\.mark\.skip/g;

+function occurrences(text: string | undefined, pattern: RegExp): number {
+  return text === undefined ? 0 : (text.match(pattern) ?? []).length;
+}
+
+export function isTestFile(path: string): boolean {
+  return TEST_FILE.test(path);
+}
+
+/**
+ * Why a test file's change looks like the tests were weakened rather than
+ * the code fixed: fewer cases or assertions than before, or new
+ * skip/only markers. `undefined` content means the file does not exist.
+ */
+export function testWeakening(before: string | undefined, after: string | undefined): string | undefined {
+  const casesBefore = occurrences(before, TEST_CASE);
+  const casesAfter = occurrences(after, TEST_CASE);
+  if (casesAfter < casesBefore) return `test cases or assertions dropped from ${casesBefore} to ${casesAfter}`;
+  if (occurrences(after, TEST_SKIP) > occurrences(before, TEST_SKIP)) return "new skip/only/todo markers";
+  return undefined;
+}

+/** True when a repo-relative path matches one of the globs. */
+export function isAllowed(repoRelativePath: string, globs: readonly string[]): boolean {
+  if (!repoRelativePath || isAbsolute(repoRelativePath)) return false;
+  const path = normalize(repoRelativePath);
+  if (path.startsWith("..")) return false;
+  return globs.some((glob) => path === normalize(glob) || matchesGlob(path, glob));

+export interface SliceFile {

+  /** Content at the base; undefined when the file is new. */
+  before: string | undefined;
+  /** Content now; undefined when the file was deleted. */
+  after: string | undefined;
+}

+/** Every reason the slice fails, from the changed files and the declared globs (empty means no file limit). */
+export function sliceFailures(files: readonly SliceFile[], globs: readonly string[]): string[] {
+  const reasons: string[] = [];
+  for (const file of files) {
+    if (globs.length > 0 && !isAllowed(file.path, globs)) reasons.push(`${file.path} is outside --files`);
